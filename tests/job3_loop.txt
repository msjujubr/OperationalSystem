// ===================================================
// Job 3: Laço de Repetição (Contagem Regressiva)
// Demonstra uso de BEQ e JUMP para controle de fluxo
// ===================================================
// Algoritmo: Contador = 3, decrementa até 0
//   R0 = Zero (constante de comparação)
//   R1 = Decremento (valor 1)
//   R3 = Contador (inicia em 3)
//
// Mapa de endereços relativos (instruções):
//   0x00: LOAD R0, [0x0A]   -> Carrega Zero = 0
//   0x01: LOAD R1, [0x0B]   -> Carrega Decremento = 1
//   0x02: LOAD R3, [0x0C]   -> Carrega Contador = 3
//   --- Início do laço (endereço 0x03) ---
//   0x03: BEQ R3, R0, 0x07  -> Se R3 == R0 (Contador==0), salta para HALT
//                               Codificação: 7308 -> Rd=3, Rf1=0, endereço=0x08
//                               NOTA: endereco = (Rf1<<4)|low = 0x08
//   0x04: SUB R3, R3, R1    -> R3 = R3 - 1
//   0x05: STORE R3, [0x0C]  -> Salva Contador atualizado na memória
//   0x06: JUMP 0x003        -> Volta ao início do laço (endereço relativo 0x03)
//   --- Fim do laço ---
//   0x07: STORE R3, [0x0C]  -> Salva resultado final (0) na memória
//   0x08: HALT              -> Fim
// ===================================================

100A // LOAD  R0, [0x0A]  (Carrega Zero = 0 em R0)
110B // LOAD  R1, [0x0B]  (Carrega Decremento = 1 em R1)
130C // LOAD  R3, [0x0C]  (Carrega Contador = 3 em R3)
7308 // BEQ   R3, R0, 0x08 (Se R3 == 0, salta para HALT no endereço 0x08)
4331 // SUB   R3, R3, R1  (R3 = R3 - 1)
330C // STORE R3, [0x0C]  (Salva Contador atualizado em [0x0C])
8003 // JUMP  0x003       (Volta para o BEQ no endereço relativo 0x03)
330C // STORE R3, [0x0C]  (Salva resultado final = 0 em [0x0C])
F000 // HALT              (Finaliza a execução do Job 3)

.data
0A 0000 // Variável Zero = 0 (constante de comparação)
0B 0001 // Variável Decremento = 1
0C 0003 // Variável Contador = 3
